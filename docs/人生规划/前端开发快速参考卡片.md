## 🎯 接到需求后的3步思考

### 第1步：这是什么功能？（5分钟）

```
□ 列表页面？ → 需要展示数据、分页、搜索
□ 表单页面？ → 需要输入数据、提交
□ 详情页面？ → 只读展示
□ 其他？ → 图表、报表等
```

### 第2步：需要哪些文件？（5分钟）

```
列表页面：
├── src/views/xxx/index.vue      # 列表页面
├── src/views/xxx/XxxForm.vue    # 表单组件（如果有）
└── src/api/xxx/index.ts         # API接口

表单页面：
├── src/views/xxx/XxxForm.vue    # 表单组件
└── src/api/xxx/index.ts         # API接口
```

### 第3步：需要哪些数据和方法？（10分钟）

```
列表页面需要：
□ list = ref([])                 # 列表数据
□ queryParams = reactive({...})   # 查询参数
□ loading = ref(false)           # 加载状态
□ total = ref(0)                 # 总条数
□ getList()                      # 获取列表方法
□ handleQuery()                  # 搜索方法
□ openForm()                     # 打开表单方法
□ handleDelete()                 # 删除方法

表单页面需要：
□ formData = reactive({...})     # 表单数据
□ formRules = {...}              # 验证规则
□ dialogVisible = ref(false)     # 对话框状态
□ formLoading = ref(false)        # 提交状态
□ open()                         # 打开表单方法
□ submitForm()                    # 提交方法
```

## 📋 代码模板

### 列表页面模板

```vue
<template>
  <!-- 搜索 -->
  <ContentWrap>
    <el-form :model="queryParams" :inline="true">
      <el-form-item label="关键词" prop="keyword">
        <el-input v-model="queryParams.keyword" @keyup.enter="handleQuery" />
      </el-form-item>
      <el-form-item>
        <el-button @click="handleQuery">搜索</el-button>
        <el-button @click="resetQuery">重置</el-button>
        <el-button type="primary" @click="openForm('create')">新增</el-button>
      </el-form-item>
    </el-form>
  </ContentWrap>

  <!-- 表格 -->
  <ContentWrap>
    <el-table :data="list" v-loading="loading">
      <el-table-column label="ID" prop="id" />
      <el-table-column label="名称" prop="name" />
      <el-table-column label="操作" width="160">
        <template #default="scope">
          <el-button @click="openForm('update', scope.row.id)">修改</el-button>
          <el-button @click="handleDelete(scope.row.id)">删除</el-button>
        </template>
      </el-table-column>
    </el-table>

    <Pagination
      :total="total"
      v-model:page="queryParams.pageNo"
      v-model:limit="queryParams.pageSize"
      @pagination="getList"
    />
  </ContentWrap>

  <XxxForm ref="formRef" @success="getList" />
</template>

<script setup lang="ts">
import * as XxxApi from '@/api/xxx'
import XxxForm from './XxxForm.vue'

const loading = ref(false)
const list = ref([])
const total = ref(0)
const formRef = ref()

const queryParams = reactive({
  pageNo: 1,
  pageSize: 10,
  keyword: undefined
})

const getList = async () => {
  loading.value = true
  try {
    const data = await XxxApi.getXxxPage(queryParams)
    list.value = data.list
    total.value = data.total
  } finally {
    loading.value = false
  }
}

const handleQuery = () => {
  queryParams.pageNo = 1
  getList()
}

const resetQuery = () => {
  Object.assign(queryParams, { pageNo: 1, pageSize: 10, keyword: undefined })
  getList()
}

const openForm = (type: string, id?: number) => {
  formRef.value?.open(type, id)
}

const handleDelete = async (id: number) => {
  try {
    await message.delConfirm()
    await XxxApi.deleteXxx(id)
    message.success('删除成功')
    await getList()
  } catch {}
}

onMounted(() => {
  getList()
})
</script>
```

### 表单组件模板

```vue
<template>
  <Dialog v-model="dialogVisible" :title="dialogTitle">
    <el-form :model="formData" :rules="formRules" ref="formRef" label-width="100px">
      <el-form-item label="名称" prop="name">
        <el-input v-model="formData.name" placeholder="请输入名称" />
      </el-form-item>
    </el-form>
    <template #footer>
      <el-button @click="submitForm">确定</el-button>
      <el-button @click="dialogVisible = false">取消</el-button>
    </template>
  </Dialog>
</template>

<script setup lang="ts">
import * as XxxApi from '@/api/xxx'

const dialogVisible = ref(false)
const formLoading = ref(false)
const formRef = ref()
const formType = ref<'create' | 'update'>('create')

const formData = reactive({
  id: undefined,
  name: ''
})

const formRules = {
  name: [{ required: true, message: '请输入名称', trigger: 'blur' }]
}

const dialogTitle = computed(() => {
  return formType.value === 'create' ? '新增' : '修改'
})

const open = async (type: 'create' | 'update', id?: number) => {
  formType.value = type
  dialogVisible.value = true
  if (type === 'update' && id) {
    await loadData(id)
  } else {
    resetForm()
  }
}

const loadData = async (id: number) => {
  formLoading.value = true
  try {
    const data = await XxxApi.getXxx(id)
    Object.assign(formData, data)
  } finally {
    formLoading.value = false
  }
}

const submitForm = async () => {
  if (!formRef.value) return
  const valid = await formRef.value.validate()
  if (!valid) return

  formLoading.value = true
  try {
    if (formType.value === 'create') {
      await XxxApi.createXxx(formData)
      message.success('创建成功')
    } else {
      await XxxApi.updateXxx(formData)
      message.success('修改成功')
    }
    dialogVisible.value = false
    emit('success')
  } finally {
    formLoading.value = false
  }
}

const resetForm = () => {
  Object.assign(formData, { id: undefined, name: '' })
  formRef.value?.resetFields()
}

const emit = defineEmits<{
  success: []
}>()

defineExpose({ open })
</script>
```

### API文件模板

```typescript
import request from '@/config/axios'

// 查询列表
export const getXxxPage = (params: any) => {
  return request.get({ url: '/xxx/page', params })
}

// 查询详情
export const getXxx = (id: number) => {
  return request.get({ url: `/xxx/get?id=${id}` })
}

// 新增
export const createXxx = (data: any) => {
  return request.post({ url: '/xxx/create', data })
}

// 修改
export const updateXxx = (data: any) => {
  return request.put({ url: '/xxx/update', data })
}

// 删除
export const deleteXxx = (id: number) => {
  return request.delete({ url: `/xxx/delete?id=${id}` })
}
```

## 🔍 常见问题速查

| 问题 | 解决方案 |
|------|---------|
| 不知道用什么组件 | 看现有代码，找类似功能 |
| 不知道数据怎么定义 | 参考现有代码的数据结构 |
| 不知道API怎么调用 | 参考现有API文件 |
| 不知道验证怎么写 | 参考现有表单的验证规则 |
| 不知道组件怎么通信 | 父→子用props，子→父用emit |

## 💡 核心原则

**前端开发 = 数据驱动视图**

1. **定义数据** → `ref()` / `reactive()`
2. **绑定数据** → `v-model` / `:prop`
3. **操作数据** → 调用API、修改数据
4. **视图更新** → Vue自动响应式更新
