#### 1.1 启动Prometheus + Grafana

```yaml
scrape_configs:

  - job_name: 'ahzx-server'
    metrics_path: '/actuator/prometheus'
    static_configs:
      - targets: ['host.docker.internal:48080']  # 开发环境
        labels:
          env: 'dev'
          app: 'ahzx-server'
    scrape_interval: 15s
    scrape_timeout: 10s

  # Redis监控（如果使用redis_exporter）
  - job_name: 'redis'
    static_configs:
      - targets: ['redis_exporter:9121']
        labels:
          env: 'dev'
```

**步骤3：验证采集**

1. 启动应用：`ahzx-server`
2. 访问：`http://localhost:48080/actuator/prometheus`，确认有指标输出
3. 访问：`http://localhost:9090`（Prometheus UI），在查询框输入 `jvm_memory_used_bytes`，确认能查到数据

**步骤4：配置Grafana Dashboard**

1. 访问：`http://localhost:3000`（Grafana，默认账号密码 admin/admin）
2. 添加数据源：Prometheus（地址：http://prometheus:9090）
3. 导入Dashboard：
   - JVM监控：导入ID `4701`（JVM Micrometer）
   - Spring Boot：导入ID `11378`（Spring Boot 2.1 Statistics）

**步骤5：配置告警规则**

创建 `monitoring-base/prometheus/alerts-basic.yml`：

```yaml
groups:
  - name: ahzx_server_alerts
    interval: 30s
    rules:
      # JVM内存告警
      - alert: HighJvmMemoryUsage
        expr: (jvm_memory_used_bytes{area="heap"} / jvm_memory_max_bytes{area="heap"}) > 0.85
        for: 5m
        labels:
          severity: warning
        annotations:
          summary: "JVM内存使用率过高"
          description: "应用 {{ $labels.instance }} 的JVM内存使用率超过85%"

      # 应用不可用告警
      - alert: ApplicationDown
        expr: up{job="ahzx-server"} == 0
        for: 1m
        labels:
          severity: critical
        annotations:
          summary: "应用不可用"
          description: "应用 {{ $labels.instance }} 已下线超过1分钟"

      # HTTP错误率告警
      - alert: HighHttpErrorRate
        expr: (sum(rate(http_server_requests_seconds_count{status=~"5.."}[5m])) / sum(rate(http_server_requests_seconds_count[5m]))) > 0.05
        for: 5m
        labels:
          severity: warning
        annotations:
          summary: "HTTP错误率过高"
          description: "应用错误率超过5%"
```

在 `prometheus.yml` 中引用：

```yaml
rule_files:
  - "alerts-basic.yml"
```

```java
@Service
public class OrderService {

        this.meterRegistry = meterRegistry;

                .tag("type", "order")
                .register(meterRegistry);

            orderCreatedCounter.increment();
            return orderId;
        });
    }
}
```

```bash
#!/bin/bash
# 慢SQL分析脚本

SLOW_LOG="/var/log/mysql/slow.log"
OUTPUT_DIR="/data/slow_sql_analysis"
DATE=$(date +%Y%m%d)

mkdir -p ${OUTPUT_DIR}

# 使用pt-query-digest分析（需要安装percona-toolkit）
if command -v pt-query-digest &> /dev/null; then
    pt-query-digest ${SLOW_LOG} > ${OUTPUT_DIR}/slow_sql_${DATE}.txt
    echo "慢SQL分析完成：${OUTPUT_DIR}/slow_sql_${DATE}.txt"
else
    echo "请安装percona-toolkit：yum install percona-toolkit"
fi
```

```java
import java.time.Duration;
import java.util.List;

/**

    /**
     * 获取缓存
     */
    <T> T get(String key, Class<T> type);
    
    /**
     * 删除缓存
     */
    void delete(String key);
    
    /**
     * 批量删除
     */
    void delete(Set<String> keys);
    
    /**
     * 判断key是否存在
     */
    Boolean exists(String key);
    
    /**
     * 设置过期时间
     */
    Boolean expire(String key, Duration timeout);
    
    /**
     * 获取剩余过期时间（秒）
     */

    /**
     * 递增
     */
```

**步骤2：实现Redis缓存服务**

创建 `ahzx-module-infra/src/main/java/com/ahzx/tdcp/module/infra/service/cache/RedisCacheServiceImpl.java`：

```java
package com.ahzx.tdcp.module.infra.service.cache;

import com.ahzx.tdcp.common.service.CacheService;
import lombok.extern.slf4j.Slf4j;
import org.springframework.data.redis.core.RedisTemplate;
import org.springframework.stereotype.Service;

import javax.annotation.Resource;
import java.time.Duration;
import java.util.Set;

@Slf4j
@Service
public class RedisCacheServiceImpl implements CacheService {
    
    @Resource
    private RedisTemplate<String, Object> redisTemplate;
    
    @Override
    public void set(String key, Object value) {
        redisTemplate.opsForValue().set(key, value);
    }
    
    @Override
    public void set(String key, Object value, Duration timeout) {
        redisTemplate.opsForValue().set(key, value, timeout);
    }
    
    @Override
    public <T> T get(String key, Class<T> type) {
        Object value = redisTemplate.opsForValue().get(key);
        if (value == null) {
            return null;
        }
        return type.cast(value);
    }
    
    @Override
    public void delete(String key) {
        redisTemplate.delete(key);
    }
    
    @Override
    public void delete(Set<String> keys) {
        redisTemplate.delete(keys);
    }
    
    @Override
    public Boolean exists(String key) {
        return redisTemplate.hasKey(key);
    }
    
    @Override
    public Boolean expire(String key, Duration timeout) {
        return redisTemplate.expire(key, timeout);
    }
    
    @Override
    public Long getExpire(String key) {
        return redisTemplate.getExpire(key);
    }
    
    @Override
    public Long increment(String key) {
        return redisTemplate.opsForValue().increment(key);
    }
    
    @Override
    public Long increment(String key, long delta) {
        return redisTemplate.opsForValue().increment(key, delta);
    }
}
```

**步骤3：业务层使用示例**

```java
@Service
public class UserService {
    
    @Autowired
    private CacheService cacheService;
    
    private static final String USER_CACHE_KEY_PREFIX = "user:";
    private static final Duration USER_CACHE_TIMEOUT = Duration.ofHours(1);
    
    public UserDO getUser(Long id) {
        // 先从缓存获取
        String cacheKey = USER_CACHE_KEY_PREFIX + id;
        UserDO user = cacheService.get(cacheKey, UserDO.class);
        if (user != null) {
            return user;
        }
        
        // 缓存未命中，查询数据库
        user = userMapper.selectById(id);
        if (user != null) {
            // 写入缓存
            cacheService.set(cacheKey, user, USER_CACHE_TIMEOUT);
        }
        return user;
    }
    
    public void updateUser(UserDO user) {
        // 更新数据库
        userMapper.updateById(user);
        // 删除缓存
        String cacheKey = USER_CACHE_KEY_PREFIX + user.getId();
        cacheService.delete(cacheKey);
    }
}
```

```java
    void publish(String topic, Object event);
    
    /**
     * 发布事件（带tag）
     */

    /**
     * 发布延迟消息
```

**步骤2：实现RocketMQ事件发布器**

创建 `ahzx-module-infra/src/main/java/com/ahzx/tdcp/module/infra/service/mq/RocketMqEventPublisher.java`：

```java
package com.ahzx.tdcp.module.infra.service.mq;

import com.ahzx.tdcp.common.service.EventPublisher;
import lombok.extern.slf4j.Slf4j;
import org.apache.rocketmq.spring.core.RocketMQTemplate;
import org.springframework.stereotype.Service;

import javax.annotation.Resource;

@Slf4j
@Service
public class RocketMqEventPublisher implements EventPublisher {
    
    @Resource
    private RocketMQTemplate rocketMQTemplate;
    
    @Override
    public void publish(String topic, Object event) {
        try {
            rocketMQTemplate.convertAndSend(topic, event);
            log.info("发布事件成功，topic: {}, event: {}", topic, event);
        } catch (Exception e) {
            log.error("发布事件失败，topic: {}, event: {}", topic, event, e);
            throw new RuntimeException("发布事件失败", e);
        }
    }
    
    @Override
    public void publish(String topic, String tag, Object event) {
        try {
            String destination = topic + ":" + tag;
            rocketMQTemplate.convertAndSend(destination, event);
            log.info("发布事件成功，topic: {}, tag: {}, event: {}", topic, tag, event);
        } catch (Exception e) {
            log.error("发布事件失败，topic: {}, tag: {}, event: {}", topic, tag, event, e);
            throw new RuntimeException("发布事件失败", e);
        }
    }
    
    @Override
    public void publishDelay(String topic, Object event, long delaySeconds) {
        try {
            // RocketMQ延迟消息，delayLevel: 1s 5s 10s 30s 1m 2m 3m 4m 5m 6m 7m 8m 9m 10m 20m 30m 1h 2h
            int delayLevel = calculateDelayLevel(delaySeconds);
            rocketMQTemplate.syncSend(topic, event, 3000, delayLevel);
            log.info("发布延迟事件成功，topic: {}, delaySeconds: {}, event: {}", topic, delaySeconds, event);
        } catch (Exception e) {
            log.error("发布延迟事件失败，topic: {}, delaySeconds: {}, event: {}", topic, delaySeconds, event, e);
            throw new RuntimeException("发布延迟事件失败", e);
        }
    }
    
    private int calculateDelayLevel(long delaySeconds) {
        // 简化实现，实际应根据RocketMQ的delayLevel规则计算
        if (delaySeconds <= 1) return 1;
        if (delaySeconds <= 5) return 2;
        if (delaySeconds <= 10) return 3;
        if (delaySeconds <= 30) return 4;
        return 5; // 默认1分钟
    }
}
```

#### 6.1 MySQL自动备份脚本

**创建 `script/backup/mysql_backup.sh`：**

```bash
#!/bin/bash
# MySQL自动备份脚本

# 配置
DB_HOST="127.0.0.1"
DB_PORT="3306"
DB_USER="root"
DB_PASSWORD="your_password"
DB_NAME="ruoyi-vue-pro"
BACKUP_DIR="/data/backup/mysql"
RETENTION_DAYS=7  # 保留7天

# 创建备份目录
mkdir -p ${BACKUP_DIR}

# 生成备份文件名
DATE=$(date +%Y%m%d_%H%M%S)
BACKUP_FILE="${BACKUP_DIR}/${DB_NAME}_${DATE}.sql"
LOG_FILE="${BACKUP_DIR}/backup.log"

# 执行备份
echo "[$(date '+%Y-%m-%d %H:%M:%S')] 开始备份数据库: ${DB_NAME}" >> ${LOG_FILE}

mysqldump -h${DB_HOST} -P${DB_PORT} -u${DB_USER} -p${DB_PASSWORD} \
    --single-transaction \
    --routines \
    --triggers \
    --events \
    --databases ${DB_NAME} \
    > ${BACKUP_FILE} 2>> ${LOG_FILE}

if [ $? -eq 0 ]; then
    # 压缩备份文件
    gzip ${BACKUP_FILE}
    BACKUP_FILE="${BACKUP_FILE}.gz"
    
    echo "[$(date '+%Y-%m-%d %H:%M:%S')] 备份成功: ${BACKUP_FILE}" >> ${LOG_FILE}
    
    # 删除过期备份
    find ${BACKUP_DIR} -name "${DB_NAME}_*.sql.gz" -mtime +${RETENTION_DAYS} -delete
    echo "[$(date '+%Y-%m-%d %H:%M:%S')] 清理${RETENTION_DAYS}天前的备份文件" >> ${LOG_FILE}
else
    echo "[$(date '+%Y-%m-%d %H:%M:%S')] 备份失败！" >> ${LOG_FILE}
    exit 1
fi
```

**配置定时任务：**

```bash
# 1. 添加执行权限
chmod +x script/backup/mysql_backup.sh

# 2. 配置crontab（每天凌晨3点执行）

0 3 * * * /path/to/script/backup/mysql_backup.sh >> /var/log/mysql_backup.log 2>&1
```

#### 6.3 Redis备份脚本（可选）

**创建 `script/backup/redis_backup.sh`：**

```bash
#!/bin/bash
# Redis备份脚本

REDIS_DIR="/data/redis"
BACKUP_DIR="/data/backup/redis"
RETENTION_DAYS=7

mkdir -p ${BACKUP_DIR}

DATE=$(date +%Y%m%d_%H%M%S)

# 备份RDB文件
if [ -f "${REDIS_DIR}/dump.rdb" ]; then
    cp ${REDIS_DIR}/dump.rdb ${BACKUP_DIR}/dump_${DATE}.rdb
    gzip ${BACKUP_DIR}/dump_${DATE}.rdb
    echo "[$(date '+%Y-%m-%d %H:%M:%S')] RDB备份成功" >> ${BACKUP_DIR}/backup.log
fi

# 备份AOF文件
if [ -f "${REDIS_DIR}/appendonly.aof" ]; then
    cp ${REDIS_DIR}/appendonly.aof ${BACKUP_DIR}/appendonly_${DATE}.aof
    gzip ${BACKUP_DIR}/appendonly_${DATE}.aof
    echo "[$(date '+%Y-%m-%d %H:%M:%S')] AOF备份成功" >> ${BACKUP_DIR}/backup.log
fi

# 清理过期备份
find ${BACKUP_DIR} -name "*.rdb.gz" -mtime +${RETENTION_DAYS} -delete
find ${BACKUP_DIR} -name "*.aof.gz" -mtime +${RETENTION_DAYS} -delete
```
